# Extract raw airline data
import pandas as pd
import sqlite3, json, datetime

def extract_all():
    conn = sqlite3.connect("skybook.db")
    flights_df   = pd.read_sql("SELECT * FROM flights", conn)
    users_df     = pd.read_sql("SELECT * FROM users", conn)
    bookings_df  = pd.read_sql("SELECT * FROM bookings", conn)
    payments_df  = pd.read_sql("SELECT * FROM payments", conn)
    conn.close()
    return flights_df, users_df, bookings_df, payments_df

T
Transform — Clean & Normalize

def transform_flights(df):
    # Handle missing values
    df["price"].fillna(df["price"].median(), inplace=True)
    df["airline"].fillna("Unknown", inplace=True)

    # Normalize prices (Min-Max scaling)
    df["price_normalized"] = (df["price"] - df["price"].min()) / \
                              (df["price"].max() - df["price"].min())

    # Standardize airport codes (uppercase, strip whitespace)
    df["origin"]      = df["origin"].str.strip().str.upper()
    df["destination"] = df["destination"].str.strip().str.upper()

    # Convert timestamps to UTC datetime
    df["departure_dt"] = pd.to_datetime(df["departure_time"], utc=True)
    df["arrival_dt"]   = pd.to_datetime(df["arrival_time"],   utc=True)

    # Derive duration in minutes
    df["duration_min"] = (df["arrival_dt"] - df["departure_dt"]) \
                           .dt.total_seconds() / 60
    return df

Load — Into Relational Database

def load_to_db(flights_clean, bookings_clean):
    conn = sqlite3.connect("skybook_warehouse.db")
    flights_clean.to_sql("dim_flights",  conn, if_exists="replace", index=False)
    bookings_clean.to_sql("fact_bookings", conn, if_exists="replace", index=False)
    conn.commit(); conn.close()
    print(f"✓ Loaded {len(flights_clean)} flights, {len(bookings_clean)} bookings")

Transform — SQL inside the database

-- Clean and transform directly with SQL
CREATE TABLE dim_flights AS
SELECT
    flight_no,
    TRIM(UPPER(origin))      AS origin,
    TRIM(UPPER(destination)) AS destination,
    -- Parse timestamp
    datetime(departure)      AS departure_dt,
    datetime(arrival)        AS arrival_dt,
    -- Cast & default missing prices
    COALESCE(CAST(price AS REAL), (
        SELECT AVG(CAST(price AS REAL)) FROM stg_flights_raw
        WHERE price IS NOT NULL
    ))                       AS price,
    -- Normalize price 0-1
    (CAST(price AS REAL) - MIN(CAST(price AS REAL)) OVER()) /
    NULLIF((MAX(CAST(price AS REAL)) OVER() -
             MIN(CAST(price AS REAL)) OVER()), 0) AS price_normalized,
    CAST(seats_total AS INT) AS seats_total,
    CAST(seats_avail AS INT) AS seats_avail,
    -- Derive duration in minutes
    CAST((julianday(arrival) - julianday(departure)) * 1440 AS INT) AS duration_min
FROM stg_flights_raw
WHERE flight_no IS NOT NULL;

Analytical Queries — Revenue & Routes

-- Most popular routes
SELECT origin || ' → ' || destination AS route,
       COUNT(*) AS bookings,
       ROUND(SUM(amount), 2) AS total_revenue
FROM fact_bookings b
JOIN dim_flights f ON b.flight_id = f.id
GROUP BY route ORDER BY bookings DESC LIMIT 10;

-- Revenue per month (trend)
SELECT strftime('%Y-%m', booked_at) AS month,
       COUNT(*) AS total_bookings,
       ROUND(SUM(amount), 2)          AS revenue
FROM fact_bookings
GROUP BY month ORDER BY month;

-- Peak booking hours
SELECT CAST(strftime('%H', booked_at) AS INT) AS hour,
       COUNT(*) AS bookings
FROM fact_bookings
GROUP BY hour ORDER BY hour;
